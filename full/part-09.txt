  function extractOpenAiText(data){
    if(typeof data?.output_text==='string') return data.output_text;
    const parts=[];
    for(const item of (data?.output||[])) for(const c of (item?.content||[])) if(typeof c?.text==='string') parts.push(c.text);
    return parts.join('\n').trim();
  }

  async function testAiConnection(){
    const {provider,key,model}=aiConfig(); if(!key||!model){alert('Informe a chave e o modelo.');return;}
    saveAiSessionConfig();setAiStatus('Testando chave, modelo e uma chamada real...','busy');
    try{
      if(provider==='openai'){
        const models=await fetchJsonOrThrow('https://api.openai.com/v1/models',{headers:{Authorization:'Bearer '+key}});
        const found=Array.isArray(models?.data)&&models.data.some(m=>m.id===model);
        const out=await fetchJsonOrThrow('https://api.openai.com/v1/responses',{method:'POST',headers:{'Content-Type':'application/json',Authorization:'Bearer '+key},body:JSON.stringify({model,input:'Responda exatamente com OK.',max_output_tokens:32})});
        const txt=extractOpenAiText(out);
        setAiStatus(`Chave válida • ${found?'modelo encontrado':'modelo respondeu'} • teste: ${txt||'OK'}`,'ok');
      }else{
        const models=await fetchJsonOrThrow('https://generativelanguage.googleapis.com/v1beta/models',{headers:{'x-goog-api-key':key}});
        const found=Array.isArray(models?.models)&&models.models.some(m=>m.name===`models/${model}`||m.name===model);
        const out=await fetchJsonOrThrow(`https://generativelanguage.googleapis.com/v1beta/models/${encodeURIComponent(model)}:generateContent`,{method:'POST',headers:{'Content-Type':'application/json','x-goog-api-key':key},body:JSON.stringify({contents:[{parts:[{text:'Responda exatamente com OK.'}]}]})});
        const txt=extractGeminiText(out);
        setAiStatus(`Chave válida • ${found?'modelo encontrado':'modelo respondeu'} • teste: ${txt||'OK'}`,'ok');
      }
    }catch(err){console.error(err);setAiStatus('Falha: '+err.message+' Se for erro de CORS/rede, use um backend/proxy seguro para a API.','err');}
  }
